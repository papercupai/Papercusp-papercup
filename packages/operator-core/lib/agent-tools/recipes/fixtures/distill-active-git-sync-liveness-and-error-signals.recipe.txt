
const optional = (value, key) => value != null && typeof value === "object" && key in value ? value[key] : null;
function routineIdentity(row) {
  const healthValue = optional(row, "health");
  const health = healthValue && typeof healthValue === "object" && !Array.isArray(healthValue) ? healthValue : {};
  const marker = health.git_sync_activity && typeof health.git_sync_activity === "object" && !Array.isArray(health.git_sync_activity)
    ? health.git_sync_activity
    : {};
  return {
    active: marker.active === true,
    phase: typeof marker.phase === "string" ? marker.phase : null,
    ownerPid: Number.isSafeInteger(marker.owner_pid) && marker.owner_pid > 0 ? marker.owner_pid : null,
    workflowId: typeof marker.workflow_id === "string" && marker.workflow_id.length > 0 ? marker.workflow_id : null
  };
}
function processMatches(rows, identity) {
  return rows.filter(row =>
    (identity.ownerPid !== null && Array.isArray(row.livePids) && row.livePids.includes(identity.ownerPid)) ||
    (identity.workflowId !== null && (row.taskId === identity.workflowId || row.rootTaskId === identity.workflowId))
  );
}

  {taskId:"workflow-probe-a",rootTaskId:"root-probe-a",livePids:[11001]},
  {taskId:"workflow-probe-b",rootTaskId:"root-probe-b",livePids:[22002]}
];
const probeA = routineIdentity({health:{git_sync_activity:{active:true,owner_pid:11001,workflow_id:"workflow-probe-a"}}});
const probeB = routineIdentity({health:{git_sync_activity:{active:true,owner_pid:22002,workflow_id:"workflow-probe-b"}}});
const pidDriftGuard =
  processMatches(probeRows,probeA).length === 1 &&
  processMatches(probeRows,probeA)[0].taskId === "workflow-probe-a" &&
  processMatches(probeRows,probeB).length === 1 &&
  processMatches(probeRows,probeB)[0].taskId === "workflow-probe-b" &&
  processMatches([{taskId:"unrelated",rootTaskId:"unrelated-root",livePids:[11001]}],probeB).length === 0;
if (!pidDriftGuard) throw new Error("git-sync routine owner PID/workflow drift guard failed");

  processInventory:{

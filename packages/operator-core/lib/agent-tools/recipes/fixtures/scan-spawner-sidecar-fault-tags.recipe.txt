const read = inputs.checkpoint || {};
const rows = Array.isArray(read.results) ? read.results : [];
const row = rows.find(value => value.id === inputs.workItemId && value.ok);
const resume = row?.resume;
if (resume?.checkpoint?.status !== "available" || resume?.nextAction?.status !== "available") {
  throw new Error("Current checkpoint boundary unavailable; no log scan performed");
}
const next = String(resume.nextAction.text || "");
const currentMarkers = [...next.matchAll(/\bNext\s+scan\s+starts\s+at\s+(@\d+(?:\.\d+)?)/gi)].map(match => match[1]);
if (currentMarkers.length > 1) throw new Error("Checkpoint next action needs exactly one current next-scan marker");
let since;
if (currentMarkers.length === 1) {
  since = currentMarkers[0];
} else {
  const boundaries = [...new Set([...next.matchAll(/\b(?:from|since|start(?:ing)?(?:\s+at)?)\s+(@\d+(?:\.\d+)?)/gi)].map(match => match[1]))];
  if (boundaries.length !== 1) throw new Error("Checkpoint next action needs exactly one explicit from/since/start @epoch boundary");
  since = boundaries[0];
}
if (inputs.since !== undefined && inputs.since !== since) throw new Error("Requested since does not match the current checkpoint boundary");
const nowSeconds = Math.floor(Date.now() / 1000);
const until = inputs.until === undefined ? "@" + nowSeconds : inputs.until;
if (!/^@\d+(?:\.\d+)?$/.test(until)) throw new Error("until must be an explicit @epoch boundary");
const startSeconds = Number(since.slice(1));
const endSeconds = Number(until.slice(1));
if (!Number.isFinite(startSeconds) || !Number.isFinite(endSeconds) || startSeconds >= endSeconds || endSeconds > nowSeconds) {
  throw new Error("Scan window must be finite, increasing, and no later than the current clock");
}

const windowOutsideJournal = r.windowOutsideJournal ?? null;

  journalError,
  windowOutsideJournal,
  unitsUnknown,
